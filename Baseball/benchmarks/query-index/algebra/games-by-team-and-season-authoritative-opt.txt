(prefix ((rdfs: <http://www.w3.org/2000/01/rdf-schema#>)
         (obo: <http://purl.obolibrary.org/obo/>)
         (cco: <https://www.commoncoreontologies.org/>)
         (base: <https://baseballontology.org/>))
  (project (?season ?team ?teamLabel ?games)
    (order (?season (desc ?games) (lcase (str ?teamLabel)))
      (extend ((?games ?.0))
        (group (?season ?team ?teamLabel) ((?.0 (count distinct ?game)))
          (graph ?graph
            (extend ((?season (year ?gameStart)))
              (sequence
                (table (vars ?teamRoleType)
                  (row [?teamRoleType base:HomeTeamRole])
                  (row [?teamRoleType base:AwayTeamRole])
                )
                (bgp
                  (triple ?game <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> base:BaseballGame)
                  (triple ?game obo:BFO_0000199 ??P0)
                  (triple ??P0 obo:BFO_0000222 ?startInstant)
                  (triple ?timestamp <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> base:BaseballTimestampICE)
                  (triple ?timestamp cco:ont00001916 ?startInstant)
                  (triple ?timestamp cco:ont00001767 ?gameStart)
                  (triple ?teamRole obo:BFO_0000197 ?team)
                  (triple ?teamRole obo:BFO_0000054 ?game)
                  (triple ?team rdfs:label ?teamLabel)
                  (triple ?teamRole <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> ?teamRoleType)
                )))))))))
