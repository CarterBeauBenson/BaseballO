(prefix ((idx: <https://w3id.org/baseball/query-index/>)
         (rdfs: <http://www.w3.org/2000/01/rdf-schema#>)
         (base: <https://baseballontology.org/>))
  (project (?season ?player ?playerLabel ?totalBases)
    (order (?season (desc ?totalBases) (lcase (str ?playerLabel)))
      (extend ((?totalBases ?.0))
        (group (?season ?player ?playerLabel) ((?.0 (sum ?bases)))
          (graph ?graph
            (sequence
              (table (vars ?hitClass ?bases)
                (row [?hitClass base:SingleProcess] [?bases 1])
                (row [?hitClass base:DoubleProcess] [?bases 2])
                (row [?hitClass base:TripleProcess] [?bases 3])
                (row [?hitClass base:HomeRunProcess] [?bases 4])
              )
              (bgp
                (triple ?result <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> idx:HitFact)
                (triple ?result idx:agent ?player)
                (triple ?result idx:game ?game)
                (triple ?result idx:hitType ?hitClass)
                (triple ?game <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> idx:GameFact)
                (triple ?game idx:season ?season)
                (triple ?player rdfs:label ?playerLabel)
              ))))))))
