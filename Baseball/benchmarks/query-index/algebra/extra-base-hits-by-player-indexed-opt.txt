(prefix ((idx: <https://w3id.org/baseball/query-index/>)
         (rdfs: <http://www.w3.org/2000/01/rdf-schema#>)
         (base: <https://baseballontology.org/>))
  (project (?player ?playerLabel ?extraBaseHits)
    (order ((desc ?extraBaseHits) (lcase (str ?playerLabel)))
      (extend ((?extraBaseHits ?.0))
        (group (?player ?playerLabel) ((?.0 (count distinct ?result)))
          (graph ?graph
            (sequence
              (table (vars ?hitClass)
                (row [?hitClass base:DoubleProcess])
                (row [?hitClass base:TripleProcess])
                (row [?hitClass base:HomeRunProcess])
              )
              (bgp
                (triple ?result <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> idx:HitFact)
                (triple ?result idx:agent ?player)
                (triple ?result idx:hitType ?hitClass)
                (triple ?player rdfs:label ?playerLabel)
              ))))))))
