         (cco: <https://www.commoncoreontologies.org/>)
         (base: <https://baseballontology.org/>))
  (project (?player ?playerLabel ?hitlessGames)
    (order ((desc ?hitlessGames) (lcase (str ?playerLabel)))

        (group (?player ?playerLabel) ((?.0 (count distinct ?game)))
          (graph ?graph
            (filter (notexists
                       (sequence

                           (triple ?hit <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> base:BaseballInstitutionalProcess)

              ))))))))
